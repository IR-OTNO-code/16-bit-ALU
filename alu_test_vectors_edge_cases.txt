# =============================================================================
# ADD  (opcode 000000) — Tests the adder with boundary and overflow conditions
# =============================================================================

# Zero identity: adding zero to zero must produce zero and set only Z.
# Confirms the zero detector and that no spurious N/C/V flags appear.
# 0x0000 + 0x0000 = 0x0000 | N=0 Z=1 C=0 V=0
0000000000000000 0000000000000000 000000 0000000000000000 0100

# Additive identity (A+0): maximum positive value plus zero must return unchanged.
# Verifies no flags are set when the result is a large but valid positive number.
# 0x7FFF + 0x0000 = 0x7FFF | N=0 Z=0 C=0 V=0
0111111111111111 0000000000000000 000000 0111111111111111 0000

# Additive identity (0+B): same as above but operands swapped.
# Confirms commutativity — the adder must treat A and B symmetrically.
# 0x0000 + 0x7FFF = 0x7FFF | N=0 Z=0 C=0 V=0
0000000000000000 0111111111111111 000000 0111111111111111 0000

# Most negative value plus zero: result stays negative, only N should be set.
# Tests that 0x8000 is correctly identified as negative without triggering C or V.
# 0x8000 + 0x0000 = 0x8000 | N=1 Z=0 C=0 V=0
1000000000000000 0000000000000000 000000 1000000000000000 1000

# All-ones plus zero: result is all-ones (0xFFFF), which is -1 in two's complement.
# Only N should be set — no carry since one operand is zero.
# 0xFFFF + 0x0000 = 0xFFFF | N=1 Z=0 C=0 V=0
1111111111111111 0000000000000000 000000 1111111111111111 1000

# All-ones plus all-ones: produces 0xFFFE with a carry out.
# Tests that C is set when unsigned overflow occurs, and N reflects the MSB.
# 0xFFFF + 0xFFFF = 0xFFFE | N=1 Z=0 C=1 V=0
1111111111111111 1111111111111111 000000 1111111111111110 1010

# Signed overflow at the positive boundary: 0x7FFF+1 wraps to 0x8000.
# pos+pos=neg is the classic V condition. N and V must both be set, C must not.
# 0x7FFF + 0x0001 = 0x8000 | N=1 Z=0 C=0 V=1
0111111111111111 0000000000000001 000000 1000000000000000 1001

# Double max_pos: 0x7FFF+0x7FFF=0xFFFE. Overflow without carry.
# Both operands are positive, result is negative — V and N set, C stays 0.
# 0x7FFF + 0x7FFF = 0xFFFE | N=1 Z=0 C=0 V=1
0111111111111111 0111111111111111 000000 1111111111111110 1001

# Double max_neg: 0x8000+0x8000 wraps all the way to zero with carry and overflow.
# neg+neg=pos is overflow. All three flags Z, C, and V must fire simultaneously.
# 0x8000 + 0x8000 = 0x0000 | N=0 Z=1 C=1 V=1
1000000000000000 1000000000000000 000000 0000000000000000 0111

# max_neg + 0xFFFF (-32768 + -1): produces 0x7FFF with carry.
# neg+neg=pos triggers V, carry out also occurs. Result is positive so N=0.
# 0x8000 + 0xFFFF = 0x7FFF | N=0 Z=0 C=1 V=1
1000000000000000 1111111111111111 000000 0111111111111111 0011

# Additive inverse pair A: 1 + (-1) = 0 with carry.
# The only correct result is Z=1 and C=1. No overflow since signs differ.
# 0x0001 + 0xFFFF = 0x0000 | N=0 Z=1 C=1 V=0
0000000000000001 1111111111111111 000000 0000000000000000 0110

# Additive inverse pair B: same values swapped to confirm commutativity of carry+zero.
# 0xFFFF + 0x0001 = 0x0000 | N=0 Z=1 C=1 V=0
1111111111111111 0000000000000001 000000 0000000000000000 0110

# Adjacent boundary values: max_pos + max_neg = 0xFFFF (-1), no overflow.
# Signs differ so V cannot be set regardless of the result sign.
# 0x7FFF + 0x8000 = 0xFFFF | N=1 Z=0 C=0 V=0
0111111111111111 1000000000000000 000000 1111111111111111 1000

# One step below max_pos: 1 + 0x7FFE = 0x7FFF. No flags at all.
# Verifies the adder does not prematurely trigger overflow near the boundary.
# 0x0001 + 0x7FFE = 0x7FFF | N=0 Z=0 C=0 V=0
0000000000000001 0111111111111110 000000 0111111111111111 0000

# Minimal positive addition: 1+1=2. Baseline sanity check with no flags.
# 0x0001 + 0x0001 = 0x0002 | N=0 Z=0 C=0 V=0
0000000000000001 0000000000000001 000000 0000000000000010 0000

# =============================================================================
# SUB  (opcode 011000) — Implemented as A + ~B + 1 (two's complement subtraction)
# C=1 means no borrow (A >= B unsigned). C=0 means borrow occurred.
# =============================================================================

# Zero minus zero: result is zero, no borrow, so both Z and C are set.
# 0x0000 - 0x0000 = 0x0000 | N=0 Z=1 C=1 V=0
0000000000000000 0000000000000000 011000 0000000000000000 0110

# max_pos minus itself: confirms self-subtraction always yields zero with no borrow.
# 0x7FFF - 0x7FFF = 0x0000 | N=0 Z=1 C=1 V=0
0111111111111111 0111111111111111 011000 0000000000000000 0110

# All-ones minus itself: same self-subtraction test for the all-ones pattern.
# 0xFFFF - 0xFFFF = 0x0000 | N=0 Z=1 C=1 V=0
1111111111111111 1111111111111111 011000 0000000000000000 0110

# max_neg minus itself: self-subtraction for the most negative value.
# 0x8000 - 0x8000 = 0x0000 | N=0 Z=1 C=1 V=0
1000000000000000 1000000000000000 011000 0000000000000000 0110

# 1 minus 0: result is 1, no borrow. Only C is set to indicate no borrow.
# 0x0001 - 0x0000 = 0x0001 | N=0 Z=0 C=1 V=0
0000000000000001 0000000000000000 011000 0000000000000001 0010

# max_pos minus 0: large positive result, no borrow. Only C set.
# 0x7FFF - 0x0000 = 0x7FFF | N=0 Z=0 C=1 V=0
0111111111111111 0000000000000000 011000 0111111111111111 0010

# 0 minus 1: produces 0xFFFF (borrow). N set, C=0 to indicate borrow. No overflow.
# 0x0000 - 0x0001 = 0xFFFF | N=1 Z=0 C=0 V=0
0000000000000000 0000000000000001 011000 1111111111111111 1000

# max_neg minus 1: neg-pos=pos is signed overflow. Result 0x7FFF, V and C set.
# 0x8000 - 0x0001 = 0x7FFF | N=0 Z=0 C=1 V=1
1000000000000000 0000000000000001 011000 0111111111111111 0011

# max_pos minus 0xFFFF (-1): pos-neg=neg is signed overflow. N and V set, no carry.
# 0x7FFF - 0xFFFF = 0x8000 | N=1 Z=0 C=0 V=1
0111111111111111 1111111111111111 011000 1000000000000000 1001

# 0 minus max_neg: subtracting the most negative number from zero.
# pos-neg=neg is overflow. Result is 0x8000, so N and V set, no carry.
# 0x0000 - 0x8000 = 0x8000 | N=1 Z=0 C=0 V=1
0000000000000000 1000000000000000 011000 1000000000000000 1001

# 0xFFFF minus 1: large negative value minus small positive. N and C set.
# 0xFFFF - 0x0001 = 0xFFFE | N=1 Z=0 C=1 V=0
1111111111111111 0000000000000001 011000 1111111111111110 1010

# 1 minus 0xFFFF: small positive minus large negative (borrow). Result is 2.
# Signs differ so no overflow. No borrow flag set since C=0, result is small positive.
# 0x0001 - 0xFFFF = 0x0002 | N=0 Z=0 C=0 V=0
0000000000000001 1111111111111111 011000 0000000000000010 0000
